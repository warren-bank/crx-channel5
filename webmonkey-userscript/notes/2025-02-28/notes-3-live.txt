const get_iso_timestamp_at_previous_half_hour_increment = function() {
  const iso_now  = (new Date()).toISOString()
  const iso_ymdh = iso_now.substring( 0, 14)
  const iso_mins = iso_now.substring(14, 16)

  const iso_timestamp = `${iso_ymdh}${ (parseInt(iso_mins, 10) >= 30) ? '30' : '00' }:00.000Z`
  return iso_timestamp
}

const get_iso_timestamp_for_date_with_offset = function(date, days, hours, minutes, seconds) {
  let offset_ms = 0
  offset_ms += seconds ? (seconds             * 1000) : 0
  offset_ms += minutes ? (minutes        * 60 * 1000) : 0
  offset_ms += hours   ? (hours     * 60 * 60 * 1000) : 0
  offset_ms += days    ? (days * 24 * 60 * 60 * 1000) : 0

  date = new Date(date)
  date = date.getTime()
  date += offset_ms
  date = new Date(date)

  return date.toISOString()
}

const get_iso_range_querystring = function() {
  const iso_timestamp_start = get_iso_timestamp_at_previous_half_hour_increment()
  const iso_timestamp_end   = get_iso_timestamp_for_date_with_offset(iso_timestamp_start, 1, 0, 0, 0)

  const querystring = `start=${ encodeURIComponent(iso_timestamp_start) }&end=${ encodeURIComponent(iso_timestamp_end) }`
  return querystring
}

const print_iso_range = function() {
